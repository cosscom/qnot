<Step>

### Define your fields

The schema is the one place you describe what can be queried. Everything else, from the UI's
editors to the types of rule values and the SQL, follows from it.

```ts twoslash title="fields.ts"
import { defineFields, f } from '@qnot/core';

export const fields = defineFields({
  name: f.text(),
  age: f.number({ min: 0, integer: true }),
  plan: f.enum(['free', 'pro', 'team']),
  verified: f.boolean(),
  signedUp: f.date({ label: 'Signed up' }),
});
```

</Step>
<Step>

### Render a builder

`<QueryBuilder>` creates a store from your fields and renders the default, unstyled UI.

```tsx title="Filters.tsx"
import '@qnot/core/styles.css';
import { QueryBuilder } from '@qnot/react';
import type { QueryOf } from '@qnot/core';
import { fields } from './fields';

export function Filters({ onChange }: { onChange: (q: QueryOf<typeof fields>) => void }) {
  return <QueryBuilder fields={fields} onChange={onChange} />;
}
```

</Step>
<Step>

### Use the query

The value is qnot's own JSON: a tree of groups and rules, typed by your schema.

```ts twoslash
import { defineFields, f, type QueryOf } from '@qnot/core';
const fields = defineFields({ age: f.number(), plan: f.enum(['free', 'pro', 'team']) });
// ---cut---
const query: QueryOf<typeof fields> = {
  kind: 'group',
  combinator: 'and',
  children: [
    { kind: 'rule', field: 'age', operator: 'gte', value: 21 },
    { kind: 'rule', field: 'plan', operator: 'in', value: ['pro', 'team'] },
  ],
};
```

Turn it into SQL, or filter rows directly:

```ts
import { compile, encode } from '@qnot/core';
import { sqlFormat } from '@qnot/core/formats/sql';

const { sql, params } = encode(sqlFormat({ dialect: 'postgres', params: 'numbered' }), query, fields);
// sql: "age >= $1 AND plan IN ($2, $3)", params: [21, 'pro', 'team']

const adults = users.filter(compile(fields, query));
```

</Step>
</Steps>

## Store-first or controlled

`<QueryBuilder>` accepts store options directly, which is enough for most forms. When other parts of
the page need the query (a preview, a results table, undo buttons), create the store yourself and
share it:

```tsx
import { QueryBuilder, QueryProvider, useHistory, useQuery, useQueryStore } from '@qnot/react';

function Page() {
  const store = useQueryStore({ fields, initial: savedQuery });
  return (
    <QueryProvider store={store}>
      <QueryBuilder store={store} />
      <Preview />
    </QueryProvider>
  );
}

function Preview() {
  const query = useQuery<typeof fields>(); // re-renders on every change
  return <pre>{JSON.stringify(query, null, 2)}</pre>;
}
```

It also works controlled, like an input. Passing back the value it gave you costs nothing, because
the store recognises its own output and skips reconciling:

```tsx
const [query, setQuery] = useState(initialQuery);
<QueryBuilder fields={fields} value={query} onChange={setQuery} />;
```
